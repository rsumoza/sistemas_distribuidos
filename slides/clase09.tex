\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}

\begin{center}
\renewcommand{\arraystretch}{1.17}
\begin{tabular}{p{0.18\textwidth}p{0.34\textwidth}p{0.36\textwidth}}
\toprule
\textbf{Concepto} & \textbf{Idea intuitiva} & \textbf{Ejemplo en Asteria Online} \\
\midrule

\end{tabular}

\end{uakeybox}

\begin{frame}[shrink=3]{Vocabulario intuitivo II: frontera y dato autoritativo}
\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}
\small
\begin{block}{Frontera}
Límite explícito que separa dos ámbitos del sistema. Indica qué responsabilidades, reglas y datos quedan de cada lado y por qué puertas se permite interactuar.
\end{block}
\begin{center}

\node[uaBoxO,text width=3.1cm] (inv) {\textbf{Inventario}\\propiedad y reserva de \#17};
\node[uaBox,text width=3.1cm,right=of inv] (market) {\textbf{Mercado}\\oferta, vendedor y precio};

\draw[uaArrowO] (inv.south east) to[bend right=18] node[below,uaTiny]{evento confirmado}(market.south west);
\end{tikzpicture}
\end{center}
\begin{uainfo}
Inventario conserva el dato autoritativo sobre el propietario. Mercado no lo modifica directamente: cruza la frontera por una operación permitida o consume un hecho ya confirmado.
\end{uainfo}
\end{frame}

\begin{frame}{Vocabulario intuitivo III: autoridad y responsabilidad}
\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}
\scriptsize
\begin{center}
\renewcommand{\arraystretch}{1.16}
\begin{tabular}{p{0.20\textwidth}p{0.35\textwidth}p{0.35\textwidth}}
\toprule
\textbf{Concepto} & \textbf{Definición de trabajo} & \textbf{En Asteria} \\
\midrule
\textbf{Autoridad} & Componente con la última palabra sobre un hecho de negocio determinado. & Inventario decide quién es el propietario válido de \texttt{EspadaSolar\#17}. \\
\textbf{Responsabilidad} (ownership) & Equipo que puede cambiar reglas y datos y que responde por métricas, incidentes y recuperación. & El equipo de Inventario mantiene el invariante de propiedad y atiende una duplicación. \\
\textbf{Dato autoritativo} & Registro reconocido como verdad oficial para decidir. & \texttt{propietario(\#17)=Sofía}; Ranking mantiene una vista derivada. \\
\bottomrule
\end{tabular}
\end{center}
\begin{uakeybox}
Autoridad responde \textbf{quién decide el hecho}; responsabilidad responde \textbf{quién sostiene esa decisión durante toda la vida del sistema}.
\end{uakeybox}
\end{frame}

\begin{frame}{Vocabulario intuitivo IV: versión y puesta en producción}
\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}

\begin{center}
\renewcommand{\arraystretch}{1.17}
\begin{tabular}{p{0.22\textwidth}p{0.35\textwidth}p{0.34\textwidth}}
\toprule
\textbf{Término} & \textbf{Definición de trabajo} & \textbf{Pregunta observable} \\
\midrule
\textbf{Unidad de publicación} & Conjunto de componentes que se publica, actualiza o retira como una misma versión. & ¿Mercado e Inventario deben salir siempre juntos? \\

\begin{uawarn}
Volver el software a una versión anterior no deshace pagos, eventos ni transferencias de propiedad ya confirmadas. Esos efectos requieren reparación de negocio.
\end{uawarn}
\end{frame}

\begin{frame}{Vocabulario intuitivo V: autonomía}
\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}
\scriptsize
\begin{block}{Definición de trabajo}
La \textbf{autonomía} es el grado en que una parte puede cambiar, publicar una versión, escalar o recuperarse sin obligar al resto a hacerlo al mismo tiempo.
\end{block}
\begin{center}
\renewcommand{\arraystretch}{1.08}
\begin{tabular}{p{0.24\textwidth}p{0.63\textwidth}}
\toprule
\textbf{Dimensión} & \textbf{Pregunta observable en Asteria} \\
\midrule
Cambio & ¿Inventario puede modificar su lógica sin exigir cambios simultáneos en Mercado? \\
Puesta en producción & ¿Ranking puede salir o volver atrás sin volver a publicar Inventario? \\
Escalado & ¿Ranking puede agregar recursos durante un torneo sin escalar también Pagos? \\
Recuperación & ¿Una falla de Ranking puede resolverse sin detener la compra? \\
\bottomrule
\end{tabular}
\end{center}
\begin{uakeybox}
Separar no crea autonomía por sí solo: una base compartida, versiones coordinadas o un equipo sin capacidad operativa pueden mantener el acoplamiento.
\end{uakeybox}
\end{frame}

\begin{frame}{Dos formas de ejecutar una secuencia de trabajo}
\UASlideTag{VOCABULARIO PREVIO}{UAIngNavy}
\small
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Cadena sincrónica}
La petición espera una respuesta antes de continuar:

\[
A\rightarrow B\rightarrow C\rightarrow D
\]

Si C tarda o falla, A todavía no puede terminar.
\end{block}
\column{0.48\textwidth}
\begin{block}{Continuación asíncrona}
Un hecho durable permite que otros continúen después:

\[
A\rightarrow log\Rightarrow\{C,D,E\}
\]

A no espera a todos los lectores.
\end{block}
\end{columns}
\begin{uainfo}
No son estilos universales. Una autorización de pago puede necesitar respuesta inmediata; actualizar ranking normalmente no.
\end{uainfo}
\end{frame}

\UASectionSlide{De una regla local\\a una coordinación entre fronteras}

\begin{frame}{Puente desde la Clase 08: ¿cuándo hace falta un protocolo?}
\UASlideTag{NÚCLEO}{UAIngNavy}
\small
\begin{block}{Regla local}
Si una sola autoridad posee el dato, puede preservar el invariante dentro de una transacción local: Inventario decide quién posee \texttt{EspadaSolar\#17}.
\end{block}
\begin{block}{Regla distribuida}
Si la decisión atraviesa procesos independientes, hace falta un \textbf{protocolo de coordinación}: una secuencia acordada de mensajes, estados durables, IDs, timeouts, retries y reglas de recuperación.
\end{block}
\begin{uakeybox}
Mover una frontera no elimina la regla de negocio: transforma una transacción local en un protocolo que debe sobrevivir fallas parciales.
\end{uakeybox}
\end{frame}

\begin{frame}{Diagnóstico de entrada: vote antes de discutir}
\UASlideTag{ACTIVIDAD}{UAIngOrange}
\small
\begin{enumerate}
\item Un monolito termina necesariamente como una maraña de dependencias.
\item Un microservicio es una clase de negocio expuesta por HTTP.
\item Separar procesos vuelve independientes a los equipos automáticamente.
\item Publicar un evento después de guardar en la base resuelve el dual write.
\item Una compensación de saga equivale a deshacer perfectamente la operación.
\end{enumerate}
\begin{block}{Consigna}
Vote V/F. No busque todavía la definición perfecta: anote qué palabra le genera desconfianza.
\end{block}
\end{frame}

\begin{frame}[t]{La escala de Asteria cambia la escena}
\UASlideTag{CASO CONDUCTOR}{UAIngOrange}
\small
\begin{block}{Supuesto para seguir la historia}
Sofía ya no compra en un mundo con pocos jugadores. Hay muchos compradores en distintas regiones, torneos que disparan consultas a Ranking y más ofertas que se publican a la vez.
\end{block}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Cambian las cargas}
Catálogo y Ranking reciben muchas lecturas; las compras modifican ofertas y reservas. Esas capacidades no crecen al mismo ritmo.
\end{block}
\column{0.48\textwidth}
\begin{block}{La regla sigue siendo la misma}
Sofía y otro jugador pueden intentar comprar \texttt{\#17} a la vez. Inventario debe decidir una única reserva válida antes de transferirla.
\end{block}
\end{columns}
\vspace{0.09cm}
\noindent\textbf{Conclusión:} más usuarios exigen escalar; separar capacidades requiere un límite medido.
\end{frame}

\begin{frame}[t,shrink=2]{Primeras respuestas sin separar capacidades}

\noindent\textbf{Criterio:} separar si estas medidas no alcanzan para cumplir el p99 de compra, aislar la carga o mejorar el tiempo hasta producción.
\end{frame}

\begin{frame}{Antes de separar: ¿qué problema concreto queremos resolver?}
\UASlideTag{CRITERIO DE DECISIÓN}{UAIngOrange}
\scriptsize
\begin{center}
\renewcommand{\arraystretch}{1.14}
\begin{tabular}{p{0.28\textwidth}p{0.34\textwidth}p{0.29\textwidth}}
\toprule
\textbf{Necesidad observable} & \textbf{Límite del diseño actual} & \textbf{Evidencia} \\
\midrule
Carga diferente & el torneo de Ranking vuelve lenta la compra & CPU, cola y p99 por capacidad \\
Ritmos de cambio & un cambio de Ranking espera la publicación global & tiempo hasta producción y coordinaciones \\
Aislamiento de fallas & una falla secundaria afecta el camino de compra & alcance del incidente \\
Seguridad o regulación & Pagos necesita permisos, auditoría y secretos específicos & incidentes y controles \\

Una \textbf{necesidad concreta de separación} es una limitación observable que la modularidad interna ya no resuelve suficientemente. Sin problema, evidencia e hipótesis de mejora, distribuir es solo agregar complejidad.
\end{uakeybox}
\end{frame}

\begin{frame}[shrink=3]{Monolito modular: una solución, no una etapa vergonzosa}
\UASlideTag{NÚCLEO}{UAIngNavy}
\small
\begin{center}

\begin{tabular}{p{0.27\textwidth}p{0.31\textwidth}p{0.32\textwidth}}
\toprule
\textbf{Dimensión} & \textbf{Monolito modular} & \textbf{Aplicación enmarañada} \\
\midrule

\end{uakeybox}

\begin{frame}{Cuándo conviene mantener Asteria modular y unida}
\UASlideTag{NÚCLEO}{UAIngNavy}
\begin{itemize}
\item El dominio todavía cambia y las fronteras no están maduras.
\item El equipo es pequeño y modifica varios módulos en conjunto.
\item La compra necesita invariantes que caben en una transacción local.
\item La carga de los componentes todavía es parecida.
\item El equipo aún no posee herramientas y guardias para operar muchos servicios.
\end{itemize}
\begin{block}{Ventaja de esperar}
Refactorizar una frontera dentro del proceso es más barato que cambiarla cuando ya es una API de red, un esquema de eventos almacenado y una responsabilidad operativa independiente.
\end{block}
\end{frame}

\begin{frame}[shrink=8]{Qué independencia puede justificar una separación}
\UASlideTag{NÚCLEO}{UAIngNavy}
\small
\begin{center}

\midrule
Publicar versiones & Ranking sale o vuelve atrás sin coordinar Mercado ni Inventario \\

\end{center}
\begin{uainfo}
Separar no crea independencia automáticamente. Una base compartida, versiones coordinadas o contratos frágiles pueden dejar el sistema igual de acoplado, pero con red en el medio.
\end{uainfo}
\end{frame}

\end{block}
\begin{columns}[T]

\end{block}
\column{0.48\textwidth}

\end{block}
\end{columns}

\noindent\textbf{Contrato:} Mercado publica \texttt{PurchaseConfirmed} con outbox e idempotencia.
\end{frame}

\begin{frame}{Disponibilidad compuesta: una aproximación útil, no una ley}
\UASlideTag{NÚCLEO}{UAIngNavy}
\small
Si cuatro dependencias independientes son obligatorias y cada una está disponible el $99.9\%$:
\[
A_{flujo}\approx 0.999^4\approx 0.996
\]
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Qué muestra}
Cada dependencia requerida puede bloquear el camino crítico.
\end{block}
\column{0.48\textwidth}
\begin{block}{Qué omite}
Fallas correlacionadas, red, colas, retries, cachés, degradación y dependencia condicional.
\end{block}
\end{columns}
\begin{uawarn}
Más procesos no implican más disponibilidad. El objetivo es retirar dependencias no críticas o permitir degradación, no multiplicar cajas.
\end{uawarn}
\end{frame}

\UASectionSlide{Cruzar la frontera:\\RPC, comandos, consultas y eventos}

\begin{frame}{Llamada local y RPC se parecen en el código, no en la falla}
\UASlideTag{NÚCLEO}{UAIngNavy}
\scriptsize
\begin{block}{RPC}
\textbf{Remote Procedure Call} o \textbf{llamada a procedimiento remoto}: contrato que hace parecer una interacción de red como una llamada a una función. Simplifica la interfaz; no elimina la incertidumbre distribuida.
\end{block}
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Llamada local}\begin{itemize}\item mismo proceso;\item error inmediato y visible;\item una pila y un entorno de ejecución;\item transacción local posible.\end{itemize}\end{block}
\column{0.48\textwidth}\begin{block}{RPC}\begin{itemize}\item serialización y red;\item petición o respuesta pueden perderse;\item el receptor puede ejecutar y el emisor observar timeout;\item requiere identidad, retry y observabilidad.\end{itemize}\end{block}
\end{columns}
\end{frame}

\begin{frame}{Intuición: un evento es un aviso durable sobre un hecho}
\UASlideTag{INTUICIÓN}{UAIngOrange}
\scriptsize
\begin{block}{Ejemplo}
\texttt{PurchaseConfirmed(purchase\_id=42, item=17, buyer=Sofía)} informa que Mercado ya confirmó la compra tras conocer la transferencia. Describe pasado confirmado; no ordena una acción futura.
\end{block}
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Qué permite}\begin{itemize}\item lectores independientes;\item replay después de una caída;\item nuevas vistas sin modificar Inventario.\end{itemize}\end{block}
\column{0.48\textwidth}\begin{block}{Qué exige}\begin{itemize}\item identidad estable;\item significado inequívoco;\item formato compatible;\item política de duplicados y orden.\end{itemize}\end{block}
\end{columns}
\begin{uainfo}
Si el mensaje pretende que un destinatario haga algo, es un comando, aunque viaje por un broker.
\end{uainfo}
\end{frame}

\begin{frame}{Contrato de evento: la interfaz pública que queda almacenada}
\UASlideTag{FORMALIZACIÓN}{UAIngNavy}
\scriptsize
\begin{block}{Definición}
El \textbf{contrato de un evento} especifica qué hecho representa, qué campos transporta, cómo se identifica, qué orden puede suponerse y cómo evoluciona sin romper consumidores.
\end{block}
\begin{center}
\renewcommand{\arraystretch}{1.10}
\begin{tabular}{p{0.21\textwidth}p{0.64\textwidth}}
\toprule
\textbf{Parte} & \textbf{En \texttt{PurchaseConfirmed}} \\
\midrule
Identidad & \texttt{event\_id}, \texttt{purchase\_id} y versión del evento \\
Semántica & Mercado confirmó la compra tras la transferencia; no es una intención pendiente \\
Esquema & nombres, tipos y obligatoriedad de campos \\
Evolución & reglas para agregar o retirar campos sin romper lectores viejos \\
Entrega & posibilidad de duplicado, orden relevante por compra y replay \\
\bottomrule
\end{tabular}
\end{center}
\begin{uakeybox}
Se parece a una API porque otros componentes dependen de esa interfaz. La diferencia es que queda almacenada: consumidores nuevos o atrasados leen eventos producidos por versiones anteriores.
\end{uakeybox}
\end{frame}

\begin{frame}{Ventajas y costos de publicar un contrato de evento}
\UASlideTag{TRADE-OFF}{UAIngOrange}
\small
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Ventajas}\begin{itemize}\item desacopla productor y lectores;\item permite replay y reconstrucción;\item agrega lectores sin cambiar Inventario;\item retira efectos secundarios del camino crítico.\end{itemize}\end{block}
\column{0.48\textwidth}\begin{block}{Costos}\begin{itemize}\item duplicados y atraso;\item compatibilidad durante años;\item observabilidad de lag y fallas;\item mala semántica se propaga a todos.\end{itemize}\end{block}
\end{columns}
\begin{uainfo}
Identidad: cuándo dos registros representan el mismo hecho. Semántica: qué afirma. Evolución: cómo cambia el formato sin romper la historia almacenada.
\end{uainfo}
\end{frame}

\begin{frame}{Dos órdenes, dos ventanas de falla}
\UASlideTag{EJECUCIÓN ADVERSARIAL}{UAIngOrange}
\scriptsize
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Base de datos primero}
\begin{enumerate}
\item confirmar compra en la base;
\item \textbf{crash};
\item evento nunca publicado.
\end{enumerate}
Resultado: compra autoritativa sin notificación ni ranking.
\end{block}
\column{0.48\textwidth}
\begin{block}{Broker primero}
\begin{enumerate}
\item publicar evento;
\item \textbf{crash o reversión de la transacción local};
\item la base no confirma la compra.
\end{enumerate}
Resultado: consumidores reaccionan a una compra inexistente.
\end{block}
\end{columns}
\begin{uakeybox}
Cambiar el orden no elimina la ventana: solo cambia qué mitad queda sola.
\end{uakeybox}
\end{frame}

\begin{frame}[shrink=2]{Intuición de outbox: dejar el aviso en la misma operación local}
\UASlideTag{INTUICIÓN}{UAIngOrange}
\small
\begin{center}
\begin{tikzpicture}[node distance=0.55cm]
\node[uaBoxO,text width=2.65cm] (tx) {Transacción local\\de Mercado};
\node[uaBox,text width=2.65cm,right=of tx,yshift=0.65cm] (state) {guardar compra\\\texttt{CONFIRMADA}};
\node[uaBox,text width=2.65cm,right=of tx,yshift=-0.65cm] (out) {guardar aviso pendiente\\en \texttt{outbox}};
\draw[uaArrow] (tx)--(state); \draw[uaArrowO] (tx)--(out);
\end{tikzpicture}
\end{center}
\begin{block}{Imagen mental}
Después de conocer la transferencia de Inventario, Mercado anota la compra confirmada y el aviso que deberá enviar. Si la transacción local confirma, existen ambos; si aborta, no existe ninguno.
\end{block}
\begin{uakeybox}
La outbox evita olvidar la intención de publicar si el proceso cae después del commit local. Todavía hace falta un proceso que publique el aviso.
\end{uakeybox}
\end{frame}

\begin{frame}{Definición formal: transactional outbox}
\UASlideTag{FORMALIZACIÓN}{UAIngNavy}
\small
\begin{block}{Definición}
Una \textbf{transactional outbox} es una tabla o colección persistente donde el servicio inserta un evento pendiente dentro de la misma transacción local que modifica el estado de negocio.
\end{block}
\begin{center}
\begin{tikzpicture}[node distance=0.48cm]
\node[uaBoxO,text width=2.35cm] (market) {Mercado\\COMMIT local};
\node[uaBox,text width=2.25cm,right=of market] (row) {fila outbox\\\texttt{PENDING}};
\node[uaBoxO,text width=2.20cm,right=of row] (relay) {Relay\\publica/reintenta};
\node[uaBox,text width=2.20cm,right=of relay] (broker) {Broker\\evento durable};
\draw[uaArrow] (market)--(row); \draw[uaArrowO] (row)--(relay); \draw[uaArrow] (relay)--(broker);
\end{tikzpicture}
\end{center}
\begin{itemize}
\item el relay puede leer por polling o CDC;
\item la publicación suele ser al menos una vez;
\item el consumidor deduplica mediante \texttt{event\_id} o inbox transaccional.
\end{itemize}
\end{frame}

\begin{frame}{Qué resuelve outbox y qué deja abierto}
\UASlideTag{TRADE-OFF}{UAIngOrange}
\small
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Resuelve}\begin{itemize}\item estado sin intención durable de publicar;\item crash entre commit local y registro del aviso;\item recuperación del trabajo pendiente.\end{itemize}\end{block}
\column{0.48\textwidth}\begin{block}{No resuelve sola}\begin{itemize}\item duplicados del relay;\item COMMIT global con Inventario o Pagos;\item consumidores no idempotentes;\item atraso y crecimiento de la tabla.\end{itemize}\end{block}
\end{columns}
\begin{uainfo}
Métricas mínimas: filas pendientes, edad de la más antigua, reintentos y tiempo extremo a extremo hasta el consumidor.
\end{uainfo}
\end{frame}

\begin{frame}[t]{Sofía ve éxito; Ranking todavía puede estar atrasado}
\UASlideTag{CASO CONDUCTOR}{UAIngOrange}
\small
\begin{enumerate}
\item Inventario confirma la transferencia. Mercado guarda \texttt{CONFIRMADA} y \texttt{PurchaseConfirmed} pendiente en un mismo COMMIT local.
\item Sofía recibe la confirmación basada en ese estado autoritativo. El relay aún puede estar detenido y Ranking mostrar la puntuación anterior.
\item El relay publica o reintenta; Ranking deduplica por \texttt{event\_id} y actualiza su proyección.
\end{enumerate}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{Beneficio}
Una falla de Ranking no borra la compra confirmada y la proyección puede recuperarse.
\end{block}
\column{0.48\textwidth}
\begin{block}{Costo observable}
Medir edad de la outbox, atraso de Ranking, duplicados y tiempo de reconstrucción.
\end{block}
\end{columns}
\end{frame}

\UASlideTag{NECESIDAD}{UAIngOrange}
\small
\begin{block}{Problema}
La compra atraviesa Mercado, Inventario y Pagos. Cada autoridad confirma localmente, pero no existe un COMMIT único que abarque a las tres.
\end{block}
\begin{center}

\end{uakeybox}
\end{frame}

\begin{frame}{Intuición: una saga es un itinerario durable de negocio}
\UASlideTag{INTUICIÓN}{UAIngOrange}
\small
\begin{block}{Imagen mental}
La compra es un viaje con varias paradas. En cada parada se anota durablemente dónde quedó, qué respuesta recibió y cuál es el próximo paso permitido.
\end{block}

\begin{uainfo}
Si el proceso reinicia, no vuelve a empezar a ciegas: lee el estado durable y continúa, consulta o repara desde allí.
\end{uainfo}
\end{frame}

\begin{frame}{Definición formal: saga}
\UASlideTag{FORMALIZACIÓN}{UAIngNavy}
\small
\begin{block}{Definición}
Una \textbf{saga} coordina una operación mediante transacciones locales. Conserva estados durables y define reintentos, consultas, timeouts y compensaciones para resultados parciales.
\end{block}
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Registra}\begin{itemize}\item \texttt{saga\_id} e intento;\item estado y deadline;\item comandos y respuestas;\item último resultado durable.\end{itemize}\end{block}
\column{0.48\textwidth}\begin{block}{Decide}\begin{itemize}\item continuar;\item consultar o reintentar;\item compensar;\item pasar a revisión manual.\end{itemize}\end{block}
\end{columns}
\end{frame}

\begin{frame}{Tres formas de ``volver atrás'' que no son equivalentes}

\begin{center}
\renewcommand{\arraystretch}{1.16}
\begin{tabular}{p{0.24\textwidth}p{0.36\textwidth}p{0.31\textwidth}}
\toprule
\textbf{Mecanismo} & \textbf{Qué hace} & \textbf{Ejemplo} \\
\midrule
Rollback transaccional & aborta cambios locales aún no confirmados & la base revierte una transacción sin COMMIT \\
Vuelta atrás de versión & devuelve tráfico o software a una versión anterior & se retira una versión defectuosa de Ranking \\
Compensación de negocio & ejecuta una nueva acción para reparar un efecto confirmado & anular autorización y liberar la reserva \\
\bottomrule
\end{tabular}
\end{center}
\begin{uawarn}
Volver una versión de software no borra pagos, eventos ni cambios de propiedad ya producidos. Esos efectos requieren acciones explícitas.
\end{uawarn}
\end{frame}

\begin{frame}{Migración gradual: patrón Strangler Fig}
\UASlideTag{VOCABULARIO JUSTO A TIEMPO}{UAIngNavy}
\small
\begin{block}{Intuición}
En vez de reemplazar toda la aplicación en una fecha única, una parte nueva crece al costado de la anterior. El tráfico se desvía por porciones hasta retirar el camino viejo.
\end{block}
\begin{block}{Definición}
El patrón \textbf{Strangler Fig} es una migración incremental donde el componente antiguo y el nuevo coexisten y una fachada decide qué camino utilizar.
\end{block}
\begin{uakeybox}
No significa simplemente ``crear un microservicio''. Es un método para reducir riesgo: coexistir, comparar, desviar y retirar.
\end{uakeybox}
\end{frame}

\begin{frame}{Prueba gradual y vuelta atrás: limitar el riesgo}
\UASlideTag{VOCABULARIO JUSTO A TIEMPO}{UAIngNavy}
\small
\begin{columns}[T]
\column{0.48\textwidth}\begin{block}{Prueba canary}Una fracción pequeña del tráfico real usa la versión nueva. Se comparan errores, latencia y resultados antes de ampliar.\end{block}
\column{0.48\textwidth}\begin{block}{Vuelta atrás de versión}Si las métricas empeoran, el router devuelve el tráfico al camino anterior. No deshace efectos de negocio ya confirmados.\end{block}
\end{columns}
\begin{uakeybox}
La migración es reversible mientras ambos caminos coexistentes puedan recibir tráfico y los datos se mantengan comparables.
\end{uakeybox}
\end{frame}
