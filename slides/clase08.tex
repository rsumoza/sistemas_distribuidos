\UASectionSlide{De la consistencia local a decisiones globales}

\begin{frame}{Puente desde clase 7}
\begin{itemize}
\item Particionar un log mejora paralelismo y elimina un orden global gratuito.
\item Replicar mejora disponibilidad, pero las copias pueden divergir.
\item Coordinar reduce ambigüedad, pero cuesta latencia y disponibilidad.
\end{itemize}
\begin{block}{Pregunta}
¿Qué errores acepta el negocio cuando una región deja de comunicarse con otra?
\end{block}
\end{frame}

\begin{frame}{Tres casos, tres costos de error}
\small
\begin{tabular}{p{0.28\textwidth}p{0.29\textwidth}p{0.32\textwidth}}
\toprule
Sistema & Error más costoso & Posible prioridad \\
\midrule
Carrito de compras & perder disponibilidad y ventas & aceptar versiones y reconciliar \\
Ledger financiero & doble gasto / saldo imposible & rechazar antes que divergir \\
Feed social & latencia alta global & aceptar frescura eventual \\
\bottomrule
\end{tabular}
\begin{uakeybox}
CAP no elige por nosotros. El dominio determina qué error es más caro.
\end{uakeybox}
\end{frame}

\UASectionSlide{CAP: dos decisiones explícitas durante una partición}

\begin{frame}{CAP sin slogan}
\begin{block}{Condición}
Existe una partición: dos subconjuntos correctos no pueden comunicarse.
\end{block}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Preservar C}
\begin{itemize}
\item rechazar o bloquear ciertas operaciones;
\item evitar historias incompatibles;
\item sacrificar disponibilidad CAP.
\end{itemize}
\column{0.48\textwidth}
\textbf{Preservar A}
\begin{itemize}
\item responder en ambos lados;
\item aceptar divergencia potencial;
\item reconciliar después.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Timeline: dos clientes, una partición}
\begin{center}
\begin{tikzpicture}[node distance=2.0cm,>=Latex,every node/.style={font=\scriptsize,align=center}]
\node[draw,rounded corners] (a) {Réplica A\\saldo=100};
\node[draw,rounded corners,right=5cm of a] (b) {Réplica B\\saldo=100};
\draw[very thick,dashed,UAIngOrange] ($(a.east)+(0.4,1.0)$)--($(a.east)+(0.4,-1.0)$);
\node[above] at ($(a.east)+(0.4,1.0)$){partición};
\node[below=0.7cm of a] (ca) {Cliente A: retirar 80};
\node[below=0.7cm of b] (cb) {Cliente B: retirar 80};
\draw[->] (ca)--(a); \draw[->] (cb)--(b);
\end{tikzpicture}
\end{center}
\begin{block}{Decisión}
Si ambos responden “éxito”, la disponibilidad se preserva y el invariante puede romperse. Si uno rechaza, se preserva corrección a costa de disponibilidad.
\end{block}
\end{frame}

\begin{frame}{Qué significa “availability” en CAP}
\small
En la formulación clásica, toda solicitud recibida por un nodo no fallado debe obtener una respuesta no errónea, sin esperar indefinidamente a la otra partición.
\begin{itemize}
\item “Tenemos réplicas” no garantiza disponibilidad CAP.
\item “Respondemos error/retry” puede significar que elegimos C.
\item Latencia alta y disponibilidad no son la misma propiedad, aunque se relacionan operacionalmente.
\end{itemize}
\end{frame}

\begin{frame}{CAP no describe la mayoría del tiempo}
\begin{itemize}
\item Las particiones son un escenario, no el estado permanente.
\item Sin partición, coordinar entre regiones sigue costando.
\item Un sistema puede elegir menor latencia aun cuando podría coordinar.
\end{itemize}
\begin{block}{Pregunta de operación normal}
¿Esperamos a una región remota para cada lectura/escritura o respondemos localmente?
\end{block}
\end{frame}

\UASectionSlide{PACELC: durante la partición y cuando todo parece sano}

\begin{frame}{PACELC}
\begin{block}{Marco}
Si hay \textbf{P}artición: elegir entre \textbf{A}vailability y \textbf{C}onsistency.\newline
\textbf{E}lse: elegir entre \textbf{L}atency y \textbf{C}onsistency.
\end{block}
\begin{itemize}
\item CAP obliga a decidir bajo partición.
\item PACELC hace visible que la coordinación también cuesta en operación normal.
\end{itemize}
\end{frame}

\begin{frame}{Ejemplo: lectura global de perfil}
\small
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Lectura local}
\begin{itemize}
\item 15 ms;
\item puede estar atrasada;
\item excelente experiencia interactiva.
\end{itemize}
\column{0.48\textwidth}
\textbf{Lectura coordinada}
\begin{itemize}
\item 180 ms interregional;
\item ve la última escritura confirmada;
\item mayor costo y cola larga.
\end{itemize}
\end{columns}
\begin{block}{Decisión}
La consistencia no es una etiqueta general: puede variar por operación y dato.
\end{block}
\end{frame}

\UASectionSlide{Dynamo: diseñar para seguir escribiendo}

\begin{frame}{Caso real: el carrito de Amazon}
\small
Dynamo fue diseñado para servicios de Amazon que requerían experiencia “always-on”. Bajo ciertas fallas, prioriza disponibilidad y acepta versiones que la aplicación puede reconciliar.
\begin{itemize}
\item No intenta ser una base relacional global.
\item Versiona objetos en lugar de ocultar todos los conflictos.
\item Distribuye responsabilidad entre infraestructura y aplicación.
\end{itemize}
{\scriptsize Fuente: DeCandia et al., “Dynamo: Amazon's Highly Available Key-value Store”.}
\end{frame}

\begin{frame}{Arquitectura Dynamo en una imagen}
\begin{center}
\begin{tikzpicture}[scale=0.95,every node/.style={font=\scriptsize,align=center},>=Latex]
\foreach \a/\n in {90/A,18/B,-54/C,-126/D,162/E}{
  \node[draw,circle,minimum size=0.75cm] (\n) at (\a:2.5cm){\n};
}
\draw[thick] (0,0) circle (2.5cm);
\node[draw,rounded corners,fill=UAIngPaper] (k) at (0,0){key $k$\\coordinator B};
\draw[->] (k)--(B); \draw[->] (B)--(C); \draw[->] (B)--(D);
\end{tikzpicture}
\end{center}
\begin{itemize}
\item Consistent hashing distribuye claves.
\item Cada clave se replica en $N$ nodos sucesores/preferencia.
\item Lecturas y escrituras esperan $R$ y $W$ respuestas.
\end{itemize}
\end{frame}

\begin{frame}{Quórum con intersección: $N=3, W=2, R=2$}
\begin{center}
\begin{tikzpicture}[node distance=1.3cm,every node/.style={font=\small,align=center}]
\node[draw,circle] (a) {A}; \node[draw,circle,right=of a] (b) {B}; \node[draw,circle,right=of b] (c) {C};
\node[below=0.7cm of b] {Escritura: A,B \qquad Lectura: B,C};
\draw[very thick,UAIngOrange] (a.south)--(b.south);
\draw[very thick,UAIngNavy] (b.north)--(c.north);
\node[above=0.6cm of b,fill=white] {intersección};
\end{tikzpicture}
\end{center}
\[
R+W>N \Rightarrow \text{intersección de conjuntos}
\]
\begin{alertblock}{Límite}
La intersección no garantiza por sí sola linearizabilidad: importan versiones, coordinación y qué réplica responde.
\end{alertblock}
\end{frame}

\begin{frame}{Sin intersección: $N=3, W=1, R=1$}
\begin{center}

\node[draw,circle] (a) {A}; \node[draw,circle,right=of a] (b) {B}; \node[draw,circle,right=of b] (c) {C};

\draw[very thick,UAIngNavy] (c.north west)--(c.north east) node[above=0.2cm]{read};
\end{tikzpicture}

\begin{itemize}
\item Escritura rápida y disponible.
\item Lectura rápida y disponible.
\item Puede leer una versión anterior sin contactar la réplica actualizada.
\end{itemize}
\end{frame}

\begin{frame}{Version vectors: distinguir “después” de “concurrente”}
\small
\begin{itemize}
\item $v_2$ domina a $v_1$ si contiene al menos todo su conocimiento y algún avance.
\item Si ninguna domina, son versiones concurrentes.
\end{itemize}
\begin{uacase}
Dos clientes agregan productos distintos al carrito durante una partición. “Último timestamp gana” puede perder uno; merge de carrito puede conservar ambos.
\end{uacase}
\end{frame}

\begin{frame}{Sloppy quorum y hinted handoff}
\begin{itemize}
\item Si un nodo preferido no responde, otro nodo temporal puede aceptar la escritura.
\item Guarda un \emph{hint}: el dato pertenece a un nodo distinto.
\item Cuando se recupera la conectividad, se hace handoff.
\end{itemize}
\begin{block}{Ganancia y costo}
Más disponibilidad de escritura; más dispersión temporal, reparación y complejidad de lectura.
\end{block}
\end{frame}

\begin{frame}{Anti-entropy y Merkle trees}
\begin{itemize}
\item Réplicas comparan resúmenes de rangos, no todos los objetos.
\item Un Merkle tree permite localizar rangos divergentes.
\item Read repair corrige durante una lectura; anti-entropy repara en background.
\end{itemize}
\begin{alertblock}{Lección}
Eventual consistency no significa “converge sola”: necesita mecanismos y capacidad de reparación.
\end{alertblock}
\end{frame}

\UASectionSlide{Spanner: pagar coordinación para una historia global fuerte}

\begin{frame}{Caso real: transacciones globales en Google}
\small
Spanner fue diseñado como una base multi-versión, globalmente distribuida y replicada sincrónicamente, con transacciones externamente consistentes.
\begin{itemize}
\item Particiona datos en grupos replicados.
\item Usa consenso para el orden dentro de cada grupo.
\item Usa TrueTime para exponer incertidumbre del reloj.
\end{itemize}
{\scriptsize Fuente: Corbett et al., “Spanner: Google's Globally-Distributed Database”.}
\end{frame}

\begin{frame}{TrueTime no entrega un instante perfecto}
\[
TT.now()=[earliest, latest]
\]
\begin{itemize}
\item El intervalo contiene la hora real según la garantía del sistema.
\item La anchura $\epsilon$ representa incertidumbre.
\item GPS, relojes atómicos y monitoreo reducen y controlan $\epsilon$.
\end{itemize}
\begin{uakeybox}
Spanner no elimina la incertidumbre del tiempo: la convierte en una API explícita.
\end{uakeybox}
\end{frame}

\begin{frame}{Commit wait: esperar para preservar orden externo}
\small
Si $T_1$ termina antes de que $T_2$ empiece, external consistency exige que $T_1$ aparezca antes en la serialización.
\begin{enumerate}
\item Asignar timestamp de commit.
\item Esperar hasta que TrueTime garantice que ese timestamp quedó en el pasado.
\item Hacer visible el commit.
\end{enumerate}
\begin{block}{Trade-off PACELC}
Incluso sin partición, más consistencia global implica latencia de coordinación y espera temporal.
\end{block}
\end{frame}

\begin{frame}{Spanner no es “un consenso global único”}
\begin{itemize}
\item Los datos se particionan en grupos; cada grupo usa consenso.
\item Transacciones multi-grupo coordinan varios participantes.
\item El diseño evita que cada operación atraviese todos los nodos del planeta.
\end{itemize}
\begin{alertblock}{Principio}
Consenso localizado y particionamiento son necesarios para escalar una garantía fuerte.
\end{alertblock}
\end{frame}

\UASectionSlide{Comparar sistemas por dominio y evidencia}

\begin{frame}{Dynamo y Spanner no intentan ganar el mismo juego}
\scriptsize
\begin{tabular}{p{0.22\textwidth}p{0.34\textwidth}p{0.34\textwidth}}
\toprule
Dimensión & Dynamo & Spanner \\
\midrule
Objetivo & always-on, baja latencia & transacciones globales fuertes \\
Partición & sigue aceptando y reconcilia & rechaza si no puede preservar garantía \\
Conflicto & versiona / merge aplicación & evita mediante orden y transacción \\
Tiempo normal & favorece respuestas locales & paga coordinación / commit wait \\
Complejidad & reconciliación y repair & infraestructura temporal y consenso \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Actividad: cuatro dominios}
\small
Asigne a cada grupo uno:
\begin{enumerate}
\item carrito de compras;
\item reserva de un único asiento;
\item feed social;
\item ledger de dinero.
\end{enumerate}
Debe definir por operación:
\begin{itemize}
\item decisión bajo partición;
\item decisión latency/consistency sin partición;
\item conflicto aceptable;
\item mecanismo de reparación o coordinación;
\item métrica/SLO.
\end{itemize}
\end{frame}

\begin{frame}{Un sistema puede mezclar modelos}
\begin{itemize}
\item Perfil: lectura local eventualmente consistente.
\item Inventario visible: cache eventual con advertencia.
\item Reserva final: transacción fuerte.
\item Analytics: log y materialized views.
\end{itemize}
\begin{block}{Arquitectura madura}
Elige la garantía mínima suficiente por dato y operación, no una etiqueta única para toda la empresa.
\end{block}
\end{frame}

\begin{frame}{Errores frecuentes}
\begin{itemize}
\item “CAP = elegir dos” sin condición de partición.
\item Confundir baja latencia con disponibilidad CAP.
\item Creer que $R+W>N$ produce automáticamente linearizabilidad.
\item Usar last-write-wins para invariantes no fusionables.
\item Tratar eventual consistency como ausencia de reparación.
\item Diseñar un consenso global único para todos los datos.
\item Comparar Dynamo y Spanner sin declarar dominio.
\end{itemize}
\end{frame}

\begin{frame}{Cierre}
\begin{block}{Idea final}
La consistencia real es una política por operación que combina costo del error, particiones, latencia, coordinación y reparación.
\end{block}
\begin{itemize}
\item Dynamo hace visible el costo de reconciliar.
\item Spanner hace visible el costo de coordinar.
\item El dominio decide qué costo es razonable.
\end{itemize}
\end{frame}

\begin{frame}{Exit ticket}
\begin{enumerate}
\item ¿Qué decisión obliga CAP durante una partición?
\item ¿Por qué $R+W>N$ no basta para afirmar linearizabilidad?
\item ¿Qué compra Spanner con commit wait y qué paga?
\end{enumerate}
\end{frame}
