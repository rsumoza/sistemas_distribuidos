\UASectionSlide{Commit Log}

\begin{frame}{Definición}
\begin{block}{}
Secuencia ordenada e inmutable: $L = (e_0, e_1, ..., e_n)$
\end{block}

\pause
\begin{itemize}
\item cada evento tiene \textbf{offset}
\item lectura secuencial
\end{itemize}
\end{frame}

\begin{frame}{Offsets}
\begin{itemize}
\item índice lógico del evento
\item monotónico por partición
\item controlado por el consumidor
\end{itemize}
\end{frame}

\UASectionSlide{Semánticas de entrega}

\begin{frame}{At-most-once}
\begin{itemize}
\item commit de offset antes de procesar
\item puede perder mensajes
\item nunca duplica
\end{itemize}
\end{frame}

\begin{frame}{At-least-once}
\begin{itemize}
\item commit después de procesar
\item puede duplicar
\item no pierde mensajes
\end{itemize}
\end{frame}

\begin{frame}{Exactly-once}
\begin{itemize}
\item no duplicación
\item no pérdida
\end{itemize}

\pause
\begin{alertblock}{Realidad}
Se logra combinando:
\begin{itemize}
\item idempotencia
\item transacciones
\item control de offsets
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Caso de diseño: decidir antes de nombrar tecnología}
\small
\begin{block}{Escenario}Un servicio crítico debe seguir operando ante latencia variable y fallas parciales. El equipo propone ``agregar más infraestructura'' sin declarar qué propiedad necesita preservar.\end{block}
\begin{enumerate}
\item ¿Cuál es el invariante o garantía que realmente importa?
\item ¿Qué falla concreta amenaza esa garantía?
\item ¿Qué mecanismo de Logs distribuidos y streaming es pertinente y cuál sería sobreingeniería?
\item ¿Qué métrica permitiría comprobar el costo de la decisión?
\end{enumerate}
\end{frame}

\UASectionSlide{Semántica extremo a extremo}
\begin{frame}{Timeline de pérdida y duplicación}
\begin{enumerate}
\item commit del offset antes del efecto + crash $\Rightarrow$ pérdida lógica;
\item efecto antes del commit del offset + crash $\Rightarrow$ repetición;
\item efecto y offset en una unidad atómica $\Rightarrow$ efecto lógico único dentro de ese límite.
\end{enumerate}
\end{frame}
\begin{frame}{Exactly-once tiene un alcance}
Kafka puede coordinar escrituras y offsets dentro de su ecosistema, pero una llamada a un sistema externo exige idempotencia, deduplicación o una transacción adicional.
\begin{alertblock}{Pregunta}
¿Exactamente una vez en el broker, en el procesador o en el efecto de negocio?
\end{alertblock}
\end{frame}
\begin{frame}{Rebalanceo y estado}
Consumer groups escalan procesamiento, pero un rebalanceo mueve particiones. El diseño debe restaurar estado, preservar offsets y evitar dos consumidores activos sobre la misma responsabilidad lógica.
\end{frame}
