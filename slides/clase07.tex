\begin{frame}{Puente desde la clase 6}
\UASlideTag{NÚCLEO}{UAIngNavy}
\begin{itemize}
\item WAL conserva historia para reconstruir \textbf{un nodo} después de un crash.
\item Un log distribuido conserva historia para coordinar \textbf{muchos consumidores} independientes.
\item En ambos casos, orden y posición permiten reconstruir progreso.
\end{itemize}
\begin{uakeybox}
Hoy la pregunta deja de ser solo ``¿sobrevive el dato?'' y pasa a ser ``¿quién leyó qué, hasta dónde avanzó y qué efecto produjo?''.
\end{uakeybox}

\begin{frame}{Diagnóstico de entrada: vote antes de discutir}
\UASlideTag{ACTIVIDAD}{UAIngOrange}
\small
\begin{enumerate}
\item Un topic de Kafka tiene un orden global único.
\item El offset identifica de manera global un evento de negocio.
\item Confirmar el offset antes de procesar evita duplicados, pero puede perder efectos.
\item Exactly-once en Kafka garantiza que un email externo se envía una sola vez.
\item Lag pequeño implica que el estado downstream es correcto.
\end{enumerate}
\begin{block}{Consigna}
Vote V/F. Al final volveremos a las cinco afirmaciones con timelines y evidencia operacional.
\end{block}
\end{frame}

\UASectionSlide{Formalizar el log: orden local,\\claves y replicación}

\begin{frame}{Modelo formal mínimo de una partición}
\UASlideTag{NÚCLEO}{UAIngNavy}
\[
P_j = (e_{j,0}, e_{j,1}, e_{j,2}, \ldots)
\]
\begin{itemize}
\item El append asigna una posición monotónica dentro de $P_j$.
\item Para $a<b$, todo consumidor de esa partición observa $e_{j,a}$ antes que $e_{j,b}$.
\item No se deriva un orden entre $e_{i,a}$ y $e_{j,b}$ cuando $i\neq j$.
\item La retención desacopla producción y consumo.
\end{itemize}
\end{frame}

\begin{frame}{La clave de partición define orden y distribución}
\UASlideTag{NÚCLEO}{UAIngNavy}
\[
partition(key) = h(key) \bmod N
\]
\begin{center}
\begin{tikzpicture}[node distance=0.55cm]
\node[uaBoxO,text width=2.0cm] (e1) {\texttt{order=9F2A}\\created};
\node[uaBoxO,text width=2.0cm,below=0.35cm of e1] (e2) {\texttt{order=9F2A}\\paid};

\begin{block}{Trade-off}
Una key estable preserva orden por entidad; una key muy popular concentra carga y crea hotspot.
\end{block}
\end{frame}

\begin{frame}{Retry del producer: dos historias indistinguibles}

\begin{frame}{Volvemos al diagnóstico inicial}
\UASlideTag{ACTIVIDAD}{UAIngOrange}
\small
\begin{enumerate}
\item \textbf{Falso:} el orden es total por partición, no por topic completo.
\item \textbf{Falso:} el offset es una posición local; la identidad de negocio es otra cosa.
\item \textbf{Verdadero:} commit antes del efecto evita replay, pero puede perder trabajo.
\item \textbf{Falso:} la garantía Kafka--Kafka no incluye automáticamente un email o API externa.
\item \textbf{Falso:} lag pequeño no prueba corrección, deduplicación ni frescura del efecto.
\end{enumerate}
\end{frame}
