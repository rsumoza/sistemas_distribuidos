\UASectionSlide{Motivación}

\begin{frame}{Del modelo lógico al físico}
Hasta ahora:
\begin{itemize}
\item transacciones
\item serializabilidad
\end{itemize}

\pause
\begin{block}{Problema}
¿Cómo implementamos esto sobre disco real?
\end{block}
\end{frame}

\begin{frame}{Restricciones físicas}
\begin{itemize}
\item escritura secuencial >> aleatoria
\item fsync es costoso
\item memoria limitada
\end{itemize}
\end{frame}

\UASectionSlide{Write-Ahead Logging}

\begin{frame}{Regla WAL}
\begin{block}{}
Nunca escribir datos persistentes antes de loguear la operación
\end{block}

\pause
\begin{itemize}
\item asegura atomicidad
\item permite recovery
\end{itemize}
\end{frame}

\begin{frame}{Ejemplo}
\begin{itemize}
\item T1: write(x=5)
\item WAL: “write x=5”
\item crash
\end{itemize}

\pause
Recovery:
\begin{itemize}
\item REDO $\Rightarrow$ reaplicar
\end{itemize}
\end{frame}

\UASectionSlide{MVCC}

\begin{frame}{Idea}
\begin{itemize}
\item múltiples versiones por registro
\item cada transacción ve snapshot
\end{itemize}
\end{frame}

\begin{frame}{Lectura}
\begin{itemize}
\item seleccionar versión con timestamp ≤ snapshot
\end{itemize}
\end{frame}

\begin{frame}{Ventaja}
\begin{itemize}
\item lectores no bloquean escritores
\item alto paralelismo
\end{itemize}
\end{frame}

\UASectionSlide{LSM Trees}

\begin{frame}{Problema de discos}
\begin{itemize}
\item random write costoso
\end{itemize}

\pause
Solución:
\begin{block}{}
convertir escrituras en secuenciales
\end{block}
\end{frame}

\begin{frame}{Arquitectura}
\begin{itemize}
\item Memtable (RAM)
\item WAL
\item SSTables (disco)

\end{frame}

\UASectionSlide{Cierre}

\begin{frame}{Idea central}
\begin{block}{}
El almacenamiento no es implementación: es diseño del sistema
\end{block}
\end{frame}


\begin{frame}{Caso de diseño: decidir antes de nombrar tecnología}
\small
\begin{block}{Escenario}Un servicio crítico debe seguir operando ante latencia variable y fallas parciales. El equipo propone ``agregar más infraestructura'' sin declarar qué propiedad necesita preservar.\end{block}
\begin{enumerate}
\item ¿Cuál es el invariante o garantía que realmente importa?
\item ¿Qué falla concreta amenaza esa garantía?
\item ¿Qué mecanismo de Sistemas de almacenamiento es pertinente y cuál sería sobreingeniería?
\item ¿Qué métrica permitiría comprobar el costo de la decisión?
\end{enumerate}
\end{frame}
\begin{frame}{Checkpoint de comprensión}
\small
Al terminar esta clase deberías poder:
\begin{itemize}
\item explicar el problema sin apoyarte en nombres de productos;
\item identificar el supuesto de falla más importante;
\item distinguir una propiedad de seguridad de una de progreso;
\item construir un escenario donde la solución deja de ser suficiente;
\item defender el trade-off ante una objeción técnica.
\end{itemize}
\end{frame}

\UASectionSlide{Decidir un motor de almacenamiento}
\begin{frame}{Caso conductor: telemetría de alta escritura}
Un sistema recibe millones de eventos diarios y debe conservarlos, consultar rangos temporales y recuperarse después de un crash.
\begin{block}{Decisión}
¿B-tree o LSM? La respuesta depende de la mezcla de lecturas, escrituras, amplificación y objetivos de recovery.
\end{block}
\end{frame}
\begin{frame}{Amplificaciones: la factura escondida}
\begin{itemize}
\item write amplification: bytes físicos por byte lógico escrito;
\item read amplification: estructuras consultadas por lectura;
\item space amplification: espacio físico respecto del estado lógico.
\end{itemize}
\begin{uakeybox}Optimizar una dimensión desplaza costo hacia otra. La compaction no es mantenimiento gratuito: es trabajo diferido.</\end{uakeybox}
\end{frame}
\begin{frame}{Checkpoint de diseño}
Para el Trabajo Final, documente: carga esperada, objetivo de recovery, patrón de consulta, política de compaction y evidencia que permitiría revisar la elección.
\end{frame}
