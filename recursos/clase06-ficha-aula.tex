\section{Leer EXPLAIN: evidencia, no decoración}

Consulta:

\begin{verbatim}
SELECT id, total, confirmed_at
FROM purchases
WHERE student_id = 8421
  AND status = 'CONFIRMED'
ORDER BY confirmed_at DESC
LIMIT 20;
\end{verbatim}

\subsection*{Plan A --- sin índice}

\begin{verbatim}
Limit
  -> Sort  (Sort Key: confirmed_at DESC)
       -> Seq Scan on purchases
            Filter: student_id=8421 AND status='CONFIRMED'
            Rows Removed by Filter: 998000
Buffers: shared hit=840
Execution Time: 13.0 ms
\end{verbatim}

\subsection*{Plan B --- con índice compuesto}

\begin{verbatim}
Limit
  -> Index Scan using idx_purchase_student_status_date
       Index Cond: student_id=8421 AND status='CONFIRMED'
Buffers: shared hit=5
Execution Time: 0.3 ms
\end{verbatim}

Complete:

\begin{enumerate}
\item El trabajo principal del Plan A fue: \rule{0.58\textwidth}{0.4pt}
\item El índice evita: \rule{0.73\textwidth}{0.4pt}
\item Tres evidencias a comparar: \rule{0.57\textwidth}{0.4pt}
\item Dos costos nuevos sobre writes: \rule{0.51\textwidth}{0.4pt}
\item Una razón por la que el tiempo puede cambiar mañana: \rule{0.35\textwidth}{0.4pt}
\end{enumerate}
